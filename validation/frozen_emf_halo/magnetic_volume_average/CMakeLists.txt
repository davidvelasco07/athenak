cmake_minimum_required(VERSION 3.16)
project(magnetic_volume_average LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Kokkos REQUIRED CONFIG)
if(NOT ATHENA_BUILD)
  message(FATAL_ERROR "Set ATHENA_BUILD to an existing AthenaK build for config.hpp")
endif()
get_filename_component(ATHENA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(check_volume_average check_volume_average.cpp)
target_include_directories(check_volume_average PRIVATE
  "${ATHENA_SOURCE}/src" "${ATHENA_BUILD}")
target_link_libraries(check_volume_average PRIVATE Kokkos::kokkos)
enable_testing()
add_test(NAME magnetic_volume_average COMMAND check_volume_average)
